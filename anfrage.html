async function loadBlockedDates() { const snap = await getDocs( collection(db, "bookings") ); bookingBlockedDates = new Set(); internalBlockedDates = new Set(); // Zuerst nur die echten Veranstaltungstage sammeln const eventDates = new Set(); snap.forEach((document) => { const booking = document.data(); // Abgelehnte Buchungen ignorieren if ( booking.bookingStatus === "abgelehnt" || booking.status === "abgelehnt" ) { return; } const ranges = (booking.ranges && booking.ranges.length) ? booking.ranges : ( booking.eventDate ? [{ start: booking.eventDate, end: booking.eventDate }] : [] ); ranges.forEach(range => { if (!range.start || !range.end) { return; } let current = dateFromIso(range.start); const end = dateFromIso(range.end); while (current <= end) { const iso = isoFromDate(current); eventDates.add(iso); current.setDate( current.getDate() + 1 ); } }); }); // Jetzt aus den echten Veranstaltungstagen // die Sperrungen berechnen eventDates.forEach(iso => { // Der eigentliche Veranstaltungstag bookingBlockedDates.add(iso); // Abholtag bookingBlockedDates.add( addDays(iso, -1) ); // Rückgabetag bookingBlockedDates.add( addDays(iso, 1) ); // 7 Tage interne Sperre NACH dem Veranstaltungstag for ( let i = 1; i <= INTERNAL_BLOCK_DAYS; i++ ) { internalBlockedDates.add( addDays(iso, i) ); } }); rebuildBlockedDates(); renderCalendar(); }
